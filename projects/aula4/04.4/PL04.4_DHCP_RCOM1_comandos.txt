4. RTR_EMPRESA
--------------------------------------------------------------------------
enable
configure terminal
! a. hostname
hostname RTR_EMPRESA
no ip domain-lookup
! b. passwords
enable secret class
line console 0
 password cisco
 login
 logging synchronous
 exit
service password-encryption
! c. banner
banner motd # Acesso nao autorizado e proibido - RTR_EMPRESA #
! interfaces
interface GigabitEthernet0/0
 description LAN 192.168.0.0/24 - SW_A2
 ip address 192.168.0.1 255.255.255.0
 no shutdown
 exit
interface GigabitEthernet0/1
 description LAN 192.168.100.0/25 - SW_A1
 ip address 192.168.100.1 255.255.255.128
 no shutdown
 exit
! d. SSH
ip domain-name istec.pt
username admin privilege 15 secret qwerty
crypto key generate rsa general-keys modulus 1024
ip ssh version 2
line vty 0 4
 login local
 transport input ssh
 exit
! e. DHCP
ip dhcp excluded-address 192.168.0.1 192.168.0.10
ip dhcp excluded-address 192.168.100.1 192.168.100.10
ip dhcp pool LAN_A2
 network 192.168.0.0 255.255.255.0
 default-router 192.168.0.1
 dns-server 8.8.8.8
 domain-name istec.pt
 exit
ip dhcp pool LAN_A1
 network 192.168.100.0 255.255.255.128
 default-router 192.168.100.1
 dns-server 8.8.8.8
 domain-name istec.pt
 exit
end
write memory

5. SW_A2 (SW_A1 igual, trocando nome e endereços)
--------------------------------------------------------------------------
enable
configure terminal
hostname SW_A2
no ip domain-lookup
enable secret class
line console 0
 password cisco
 login
 logging synchronous
 exit
service password-encryption
banner motd # Acesso nao autorizado e proibido - SW_A2 #
interface vlan 1
 ip address 192.168.0.2 255.255.255.0
 no shutdown
 exit
ip default-gateway 192.168.0.1
ip domain-name istec.pt
username admin privilege 15 secret qwerty
crypto key generate rsa general-keys modulus 1024
ip ssh version 2
line vty 0 15
 login local
 transport input ssh
 exit
end
write memory

SW_A1: hostname SW_A1 | interface vlan 1: 192.168.100.2 255.255.255.128
       ip default-gateway 192.168.100.1

12. VERIFICAÇÃO
--------------------------------------------------------------------------
Routers:  show ip dhcp binding      -> endereços atribuídos
          show ip dhcp pool         -> estado das pools
          show ip route             -> rotas S (estáticas) e C (ligadas)
          show ip ssh               -> SSH Enabled - version 2.0
PCs:      ipconfig /all             -> IP obtido por DHCP, gateway e DNS
          ping entre PCs de redes diferentes (ex.2: SEDE <-> PROD/CONTAB)
          ssh -l admin 192.168.0.1  -> password qwerty
